# Functions for ode()
 # w for weak
 mut_w <- function(time, state, pars) {
   with(as.list(c(state, pars)), {
     FR1 <- alpha12*N2
     FR2 <- alpha21*N1
     dN1 <- r1*N1*(K1 - N1 + FR1)/K1
     dN2 <- r2*N2*(K2 - N2 + FR2)/K2
     return(list(c(dN1, dN2)))
   })
 }
 
 # s for strong
 mut_s <- function(time, state, pars) {
   with(as.list(c(state, pars)), {
     FR1 <- alpha12*N2/(h1 + N2)
     FR2 <- alpha21*N1/(h2 + N1)
     dN1 <- r1*N1*(K1 - N1 + FR1)/K1
     dN2 <- r2*N2*(K2 - N2 + FR2)/K2
     return(list(c(dN1, dN2)))
   })
 }

# Write a function for both the nullclines and vector field
 phasePlane <- function(lims, parm_vec, funky) {
   plot(x = NA, type = "n", xlim = lims, ylim = lims, las = 1, xlab = "N1", ylab = "N2")
   trash <- flowField(deriv = funky, xlim = lims, ylim = lims, parameters = parm_vec, add = T, state.names = c("N1", "N2"), points = 8, arrow.type = "proportional")
   trash <- nullclines(deriv = funky, xlim = lims + c(-0.1, 0.1), ylim = lims + c(-0.1, 0.1), parameters = parm_vec, add = T, state.names = c("N1", "N2"), col = c("#ef3b24", "#007ac1"), add.legend = F, lwd = 1.5, points = 200)
 }

MR <- function(t, y, parms) {
 with(as.list(c(y, parms)), {
   dVdt <- r*V*(K-V)/K - c*V*P/(g + V)
   dPdt <- b*V*P/(g + V) - m*P
   return(list(c(dVdt, dPdt)))
 })
}

parm_vec_eq <- c(r = 0.5, K = 15, c = 2, g = 10, b = 1.5, m = 1)
parm_vec_stab <- c(r = 0.5, K = 30, c = 2, g = 10, b = 1.5, m = 1)
